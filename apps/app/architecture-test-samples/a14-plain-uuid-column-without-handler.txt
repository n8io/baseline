import { index, pgSchema, text, uuid } from "drizzle-orm/pg-core";

export const tradeSchema = pgSchema("trade");

export const trade = tradeSchema.table(
	"trade",
	{
		id: uuid().primaryKey(),
		proposerTeamId: uuid("proposer_team_id").notNull(),
		status: text().notNull(),
	},
	(t) => [index("trade_proposer_team_id_idx").on(t.proposerTeamId)],
);
