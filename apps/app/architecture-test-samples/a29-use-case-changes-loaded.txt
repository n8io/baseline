// A29: a Use case never changes a loaded Model in place or copies it under a short name
// expect: src/contexts/billing/application/invoice-line-rename.ts:4: line.name = makes or changes a InvoiceLine in a Use case
// @file src/contexts/billing/application/invoice-line-rename.ts
export const invoiceLineRename = (ports) => (actor, input) =>
	ports.billingTransaction(async (t) => {
		const line = await t.invoiceLineRepository.findByIdForUpdate(input.id);
		line.name = input.name;
		return t.invoiceLineRepository.update(line);
	});
// @file src/contexts/billing/domain/invoice-line.ts
export type InvoiceLine = { id: string; name: string };
// @file src/contexts/billing/adapters/outbound/invoice-line-table.ts
export const invoiceLine = pgSchema("billing").table("invoice_line", {});
