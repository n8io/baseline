import { ok } from "@n8io/baseline-result";

export const barGet = (ports: { barRepository: { get(id: string): Promise<{ id: string } | undefined> } }) =>
	async (barId: string) => {
		const bar = await ports.barRepository.get(barId);
		if (!bar) throw new Error(`Bar ${barId} not found`);
		return ok(bar);
	};
